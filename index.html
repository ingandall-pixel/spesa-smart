async function salvaDatiRemoti(tentativi = 3) { mostraNotificaSync("Salvataggio su GitHub..."); for (let i = 0; i < tentativi; i++) { try { // 1. Leggi sempre l'ultimo sha disponibile prima di scrivere let sha = null; const existing = await githubApiRequest('data.json', 'GET'); if (existing && existing.sha) { sha = existing.sha; } // Nota: Se vuoi unire i dati remoti con quelli locali in caso di modifiche concorrenti, // potresti fare un merge qui. Per ora ricarichiamo la versione più recente della spesa/buoni // per evitare di sovrascrivere i dati inseriti dall'altro utente. if (existing && existing.content) { const jsonStringRemoto = decodeURIComponent(escape(atob(existing.content.replace(/\s/g, '')))); const dbRemoto = JSON.parse(jsonStringRemoto); // Unione intelligente (opzionale ma consigliata): // Uniamo gli array di spesa e buoni per non perdere i dati inseriti da un altro utente nei secondi scorsi. // Usiamo un Set o controlliamo gli ID per evitare doppioni. db.spesa = unisciListeSpesa(db.spesa, dbRemoto.spesa); db.buoni = unisciListeBuoni(db.buoni, dbRemoto.buoni); } const jsonString = JSON.stringify(db, null, 2); const contentBase64 = btoa(unescape(encodeURIComponent(jsonString))); const body = { message: "Aggiornamento dati Spesa Smart", content: contentBase64, branch: BRANCH }; if (sha) body.sha = sha; await githubApiRequest('data.json', 'PUT', body); nascondiNotificaSync(); return; // Salvataggio riuscito con successo } catch (e) { console.warn(`Tentativo di salvataggio ${i + 1} fallito:`, e); if (i === tentativi - 1) { nascondiNotificaSync(); alert("Errore di sincronizzazione con GitHub dopo diversi tentativi. Qualcun altro sta modificando i dati contemporaneamente, riprova tra poco."); } else { // Attendi un secondo prima di riprovare await new Promise(resolve => setTimeout(resolve, 1000)); } } } }
